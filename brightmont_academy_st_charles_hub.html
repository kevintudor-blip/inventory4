Google Apps Script Bridge Code

Paste this code in Google Sheets (Extensions > Apps Script) and deploy as a Web App (Execute as: Me, Access: Anyone) to write back data in real-time. 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.openById('1AVOthk_4lPCPi1zroepEjb19DozSYNyfwP2eifHjNr0').getSheetByName('Inventory') || SpreadsheetApp.openById('1AVOthk_4lPCPi1zroepEjb19DozSYNyfwP2eifHjNr0').getSheets()[0];
    var data = JSON.parse(e.postData.contents);
    sheet.clear();
    
    var headers = [
      "Primary Subject Area",
      "Ordering Name and/or link, if it applies",
      "Description",
      "Quantity",
      "Rooms/Space Storing Them",
      "Contact people",
      "Consumable? Answer \"Yes\" or, if no, leave blank. If \"Yes,\" inform Meredith so that she can replace it.",
      "Other notes"
    ];
    sheet.appendRow(headers);
    
    data.forEach(function(item) {
      sheet.appendRow([
        item.subject || '',
        item.name || '',
        item.description || '',
        item.quantity || 0,
        item.location || '',
        item.contact || '',
        (item.isConsumable === 'Yes' || item.isConsumable === true) ? 'Yes' : '',
        item.notes || ''
      ]);
    });
    
    return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
